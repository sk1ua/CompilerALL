int global_x = 100;

int add(int a, int b) {
  int result;
  result = a + b;
  return result;
}

int factorial(int n) {
  int res;
  int i = 1;
  res = 1;
  do {
    res = res * i;
    i = i + 1;
  } while (i <= n);
  return res;
}

main() {
  int x = 10, y = 5, z;
  int a[3];
  int i = 0;
  int sum;

  x = global_x;
  a[0] = 1;
  a[1] = 2;
  a[2] = 3;

  while (i < 3) {
    sum = sum + a[i];
    i = i + 1;
  }

  write sum;

  z = add(x, y);
  write z;

  z = factorial(4);
  write z;

  switch (y) {
    case 1:
      write 111;
      break;
    case 5:
      write 555;
      break;
    default:
      write 999;
  }

  do {
    y = y - 1;
    write y;
  } while (y > 0);

  return;
}
